223static WstStkErr stk_prepare_push(STK_T *stk)
224{
225 assert(stk != NULL);
226
227 WstStkErr err = stk_verify(stk);
228 if (err != WST_STK_NO_ERR || stk->length < stk->cap)
229 return err;
230
231 size_t limit = (SIZE_MAX - STK_CANARY_TOTAL) / sizeof(STK_ELEM);
232 size_t cap = (stk->cap == 0) ? STK_INIT_CAP : stk->cap * 2;
233 if (cap < stk->cap || cap > limit)
234 cap = limit;
235
236 err = stk_reserve(stk, cap);
237 if (err != WST_STK_NO_ERR)
238 return err;
239
240 if (stk->length == stk->cap) {
241 LOG_E("cannot grow past %zu elements", stk->cap);
242 return WST_STK_ERR_OVERFLOW;
243 }
244 return WST_STK_NO_ERR;
245}

280WstStkErr stk_pop(STK_T *stk, STK_ELEM *out)
281{
282 assert(stk != NULL);
283
284 WstStkErr err = stk_verify(stk);
285 if (err != WST_STK_NO_ERR)
286 return err;
287
288 if (stk->length == 0) {
289 LOG_E("pop from an empty stack");
290 return WST_STK_ERR_EMPTY;
291 }
292
293#ifdef STK_REF
294
295 stk->length--;
296 if (out != NULL)
297 *out = stk->data[stk->length];
298 wst_stk_void_seal((WstStkVoid *)stk, sizeof(STK_ELEM));
299 LOG_V("pop [%p], length %zu", (const void *)out, stk->length);
300
301#else /* !STK_REF */
302
303 STK_ELEM value = stk->data[--stk->length];
304 if (out != NULL)
305 *out = value;
306 wst_stk_void_seal((WstStkVoid *)stk, sizeof(STK_ELEM));
307 LOG_V("pop " STK_FMT ", length %zu", value, stk->length);
308
309#endif /* !STK_REF */
310
311 return WST_STK_NO_ERR;
312}
