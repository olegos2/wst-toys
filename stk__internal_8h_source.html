189static WstStkErr stk_prepare_push(STK_T *stk)
190{
191 assert(stk != NULL);
192
193 WstStkErr err = stk_verify(stk);
194 if (err != WST_STK_NO_ERR || stk->length < stk->cap)
195 return err;
196
197 size_t limit = (SIZE_MAX - STK_CANARY_SIZE) / sizeof(STK_ELEM);
198 size_t cap = (stk->cap == 0) ? STK_INIT_CAP : stk->cap * 2;
199 if (cap < stk->cap || cap > limit)
200 cap = limit;
201
202 err = stk_reserve(stk, cap);
203 if (err != WST_STK_NO_ERR)
204 return err;
205
206 if (stk->length == stk->cap) {
207 LOG_E("cannot grow past %zu elements", stk->cap);
208 return WST_STK_ERR_OVERFLOW;
209 }
210 return WST_STK_NO_ERR;
211}

246WstStkErr stk_pop(STK_T *stk, STK_ELEM *out)
247{
248 assert(stk != NULL);
249
250 WstStkErr err = stk_verify(stk);
251 if (err != WST_STK_NO_ERR)
252 return err;
253
254 if (stk->length == 0) {
255 LOG_E("pop from an empty stack");
256 return WST_STK_ERR_EMPTY;
257 }
258
259#ifdef STK_REF
260
261 stk->length--;
262 if (out != NULL)
263 *out = stk->data[stk->length];
264 wst_stk_void_seal((WstStkVoid *)stk);
265 LOG_V("pop [%p], length %zu", (const void *)out, stk->length);
266
267#else /* !STK_REF */
268
269 STK_ELEM value = stk->data[--stk->length];
270 if (out != NULL)
271 *out = value;
272 wst_stk_void_seal((WstStkVoid *)stk);
273 LOG_V("pop " STK_FMT ", length %zu", value, stk->length);
274
275#endif /* !STK_REF */
276
277 return WST_STK_NO_ERR;
