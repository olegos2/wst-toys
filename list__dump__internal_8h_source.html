14static size_t list_dump_total(const LIST_T *list)
15{
16 size_t total = 0;
17 for (size_t i = 0; i < list->blocks.length; i++) {
18 const LIST_BLOCK_T *block = list->blocks.data[i];
19 if (block != NULL)
20 total += block->cap;
21 }

26static bool list_dump_slot_of(const LIST_T *list, const void *ptr,
27 size_t *out)
28{
29 size_t base = 0;
30
31 for (size_t i = 0; i < list->blocks.length; i++) {
32 const LIST_BLOCK_T *block = list->blocks.data[i];
33 if (block == NULL)
34 continue;
35
36 uintptr_t lo = (uintptr_t)&block->nodes[0];
37 uintptr_t hi = (uintptr_t)&block->nodes[block->cap];
38 uintptr_t addr = (uintptr_t)ptr;
39
40 if (addr >= lo && addr < hi) {
41 size_t off = (size_t)(addr - lo);
42 if (off % sizeof(LIST_NODE_T) != 0)
43 return false;
44 *out = base + off / sizeof(LIST_NODE_T);
45 return true;
46 }
47 base += block->cap;
48 }
49
50 return false;
51}

54static const LIST_NODE_T *list_dump_node_at(const LIST_T *list, size_t slot)
55{
56 for (size_t i = 0; i < list->blocks.length; i++) {
57 const LIST_BLOCK_T *block = list->blocks.data[i];
58 if (block == NULL)
59 continue;
60 if (slot < block->cap)
61 return &block->nodes[slot];
62 slot -= block->cap;
63 }
64 return NULL;
65}

81static void list_dump_mark_live(const LIST_T *list, WstListSlotState *state,
82 size_t total)
83{
84 const WstListNodeHdr *node = list->hdr.head;
85 for (size_t step = 0; node != NULL && step < total; step++) {
86 size_t slot;
87 if (!list_dump_slot_of(list, node, &slot))
88 break;
89 if (state[slot] == WST_LIST_SLOT_ORPHAN)
90 state[slot] = WST_LIST_SLOT_LIVE;
91 node = node->next;
92 }
93}

110static void list_dump_check_chain(const LIST_T *list, const void *start,
111 const char *name, size_t total,
112 WstListDiag *diag)
113{
114 const WstListNodeHdr *node = start;
115 for (size_t step = 0; node != NULL; step++) {
116 size_t slot;
117 if (!list_dump_slot_of(list, node, &slot) || slot >= total) {
118 wst_list_diag_add(diag, "%s chain leaves node blocks at %p", name,
119 (const void *)node);
120 return;
121 }
122 if (step >= total) {
123 wst_list_diag_add(diag, "%s chain does not terminate", name);
124 return;
125 }
126 node = node->next;
127 }
128}

131static void list_dump_check_live(const LIST_T *list, size_t slot,
132 const WstListNodeHdr *node,
133 const WstListSlotState *state, size_t total,
134 bool head_ok, size_t head_slot,
135 bool tail_ok, size_t tail_slot,
136 WstListDiag *diag)
137{
138 const WstListNodeHdr *links[] = { node->prev, node->next };
139 const char *names[] = { "prev", "next" };
140
141 for (int i = 0; i < 2; i++) {
142 const WstListNodeHdr *target = links[i];
143 if (target == NULL) {
144 if (i == 0 && (!head_ok || slot != head_slot))
145 wst_list_diag_add(
146 diag, "live slot %zu prev is NULL but is not head", slot);
147 if (i == 1 && (!tail_ok || slot != tail_slot))
148 wst_list_diag_add(
149 diag, "live slot %zu next is NULL but is not tail", slot);
150 continue;
151 }
152
153 size_t target_slot;
154 if (!list_dump_slot_of(list, target, &target_slot) ||
155 target_slot >= total) {
156 wst_list_diag_add(
157 diag, "live slot %zu %s=%p is outside node blocks", slot,
158 names[i], (const void *)target);
159 continue;
160 }
161 if (state[target_slot] != WST_LIST_SLOT_LIVE) {
162 wst_list_diag_add(diag, "live slot %zu %s -> %s slot %zu", slot,
163 names[i],
164 wst_list_slot_state_name(state[target_slot]),
165 target_slot);
166 continue;
167 }
168
169 const LIST_NODE_T *target_node =
170 list_dump_node_at(list, target_slot);
171 if (target_node == NULL)
172 continue;
173
174 const WstListNodeHdr *back =
175 i == 0 ? target_node->hdr.next : target_node->hdr.prev;
176 if (back != node)
177 wst_list_diag_add(diag,
178 "live slot %zu %s=%zu but slot %zu %s=%p",
179 slot, names[i], target_slot, target_slot,
180 i == 0 ? "next" : "prev", (const void *)back);
181 }
182}

186 const WstListNodeHdr *node,
187 const WstListSlotState *state, size_t total,
188 WstListDiag *diag)
189{
190 if (node->prev != NULL)
191 wst_list_diag_add(diag, "free slot %zu prev=%p should be NULL", slot,
192 (const void *)node->prev);
193
194 if (node->next == NULL)
195 return;
196
197 size_t target_slot;
198 if (!list_dump_slot_of(list, node->next, &target_slot) ||
199 target_slot >= total) {
200 wst_list_diag_add(diag,
201 "free slot %zu next=%p is outside node blocks",
202 slot, (const void *)node->next);
203 } else if (state[target_slot] != WST_LIST_SLOT_FREE) {
204 wst_list_diag_add(diag, "free slot %zu next -> %s slot %zu", slot,
205 wst_list_slot_state_name(state[target_slot]),
206 target_slot);
207 }

211static void list_dump_check(const LIST_T *list, const WstListSlotState *state,
212 size_t total, WstListDiag *diag)
213{
214 if (state == NULL) {
215 wst_list_diag_add(diag, "slot state allocation failed");
216 return;
217 }
218
219 for (size_t i = 0; i < list->blocks.length; i++) {
220 const LIST_BLOCK_T *block = list->blocks.data[i];
221 if (block == NULL)
222 wst_list_diag_add(diag, "blocks[%zu] is NULL", i);
223 else if (block->cap == 0)
224 wst_list_diag_add(diag, "block[%zu] has cap 0", i);
225 }
226
227 size_t head_slot = 0;
228 size_t tail_slot = 0;
229 size_t free_slot = 0;
230 bool head_ok = list->hdr.head != NULL &&
231 list_dump_slot_of(list, list->hdr.head, &head_slot) &&
232 head_slot < total;
233 bool tail_ok = list->hdr.tail != NULL &&
234 list_dump_slot_of(list, list->hdr.tail, &tail_slot) &&
235 tail_slot < total;
236 bool free_ok = list->hdr.free != NULL &&
237 list_dump_slot_of(list, list->hdr.free, &free_slot) &&
238 free_slot < total;
239
240 if (list->hdr.head != NULL && !head_ok)
241 wst_list_diag_add(diag, "head=%p is outside node blocks",
242 (const void *)list->hdr.head);
243 if (list->hdr.tail != NULL && !tail_ok)
244 wst_list_diag_add(diag, "tail=%p is outside node blocks",
245 (const void *)list->hdr.tail);
246 if (list->hdr.free != NULL && !free_ok)
247 wst_list_diag_add(diag, "free=%p is outside node blocks",
248 (const void *)list->hdr.free);
249 if (list->hdr.head == NULL && list->hdr.tail != NULL)
250 wst_list_diag_add(diag, "head is NULL but tail=%p",
251 (const void *)list->hdr.tail);
252 if (list->hdr.head != NULL && list->hdr.tail == NULL)
253 wst_list_diag_add(diag, "tail is NULL but head=%p",
254 (const void *)list->hdr.head);
255
256 if (head_ok && state[head_slot] != WST_LIST_SLOT_LIVE)
257 wst_list_diag_add(diag, "head -> %s slot %zu",
258 wst_list_slot_state_name(state[head_slot]),
259 head_slot);
260 if (tail_ok && state[tail_slot] != WST_LIST_SLOT_LIVE)
261 wst_list_diag_add(diag, "tail -> %s slot %zu",
262 wst_list_slot_state_name(state[tail_slot]),
263 tail_slot);
264 if (free_ok && state[free_slot] != WST_LIST_SLOT_FREE)
265 wst_list_diag_add(diag, "free -> %s slot %zu",
266 wst_list_slot_state_name(state[free_slot]),
267 free_slot);
268
269 if (head_ok && list->hdr.head->prev != NULL)
270 wst_list_diag_add(diag, "head slot %zu prev=%p should be NULL",
271 head_slot, (const void *)list->hdr.head->prev);
272 if (tail_ok && list->hdr.tail->next != NULL)
273 wst_list_diag_add(diag, "tail slot %zu next=%p should be NULL",
274 tail_slot, (const void *)list->hdr.tail->next);
275 if (free_ok && list->hdr.free->prev != NULL)
276 wst_list_diag_add(diag, "free head slot %zu prev=%p should be NULL",
277 free_slot, (const void *)list->hdr.free->prev);
278
279 list_dump_check_chain(list, list->hdr.head, "live", total, diag);
280 list_dump_check_chain(list, list->hdr.free, "free", total, diag);
281
282 for (size_t slot = 0; slot < total; slot++) {
283 const LIST_NODE_T *node = list_dump_node_at(list, slot);
284 if (node == NULL) {
285 wst_list_diag_add(diag, "slot %zu has no block", slot);
286 continue;
287 }
288
289 switch (state[slot]) {
290 case WST_LIST_SLOT_LIVE:
291 list_dump_check_live(list, slot, &node->hdr, state, total,
292 head_ok, head_slot, tail_ok, tail_slot,
293 diag);
294 break;
295 case WST_LIST_SLOT_FREE:
296 list_dump_check_free(list, slot, &node->hdr, state, total, diag);
297 break;
298 case WST_LIST_SLOT_ORPHAN:
299 default:
300 wst_list_diag_add(diag, "slot %zu is neither live nor free", slot);
301 break;
302 }
303 }
304}

306static void list_dump_graph(const LIST_T *list, size_t issues)
307{
308 if (!wst_log_accepts_images())
309 return;
310
311 size_t total = list_dump_total(list);
312 WstListSlotState *state =
313 calloc(total > 0 ? total : 1, sizeof(*state));
314 if (state == NULL)
315 return;
316
317 list_dump_mark_live(list, state, total);
318 list_dump_mark_free(list, state, total);
319 WstListSlotCounts counts = wst_list_slot_counts(state, total);
320
321 char free_ref[32];
322 list_dump_ref(list, list->hdr.free, free_ref, sizeof(free_ref));
323
324 WstListGraph *graph = wst_list_graph_new(
325 WST_LIST_STR(LIST_T),
326 "blocks=%zu\nslots=%zu\nlive=%zu\npooled=%zu\n"
327 "orphan=%zu\nfree=%s\nissues=%zu",
328 list->blocks.length, total, counts.live, counts.pooled,
329 counts.orphan, free_ref, issues);
330 if (graph == NULL) {
331 free(state);
332 return;
333 }
334
335 size_t base = 0;
336 for (size_t b = 0; b < list->blocks.length; b++) {
337 const LIST_BLOCK_T *block = list->blocks.data[b];
338 if (block == NULL)
339 continue;
340
341 wst_list_graph_block_begin(graph, b, block->cap, block);
342 for (size_t j = 0; j < block->cap; j++) {
343 size_t slot = base + j;
344 const LIST_NODE_T *node = &block->nodes[j];
345 WstListSlotState node_state = state[slot];
346
347 char prev_ref[32];
348 char next_ref[32];
349 list_dump_ref(list, node->hdr.prev, prev_ref, sizeof(prev_ref));
350 list_dump_ref(list, node->hdr.next, next_ref, sizeof(next_ref));
351
352#ifdef LIST_FMT
353 wst_list_graph_node(graph, slot, node_state,
354 "#%zu\nprev=%s\nnext=%s\ndata=" LIST_FMT,
355 slot, prev_ref, next_ref, node->data);
356#else
357 wst_list_graph_node(graph, slot, node_state,
358 "#%zu\nprev=%s\nnext=%s\ndata=%p", slot,
359 prev_ref, next_ref, (const void *)&node->data);
360#endif
361 }
362 wst_list_graph_block_end(graph);
363 base += block->cap;
364 }
365
366 /* Define every node before its edges so cross-block links do not create
367 the target inside the previous block's cluster. */
368 base = 0;
369 for (size_t b = 0; b < list->blocks.length; b++) {
370 const LIST_BLOCK_T *block = list->blocks.data[b];
371 if (block == NULL)
372 continue;
373
374 wst_list_graph_block_begin(graph, b, block->cap, block);
375 for (size_t j = 0; j < block->cap; j++) {
376 size_t slot = base + j;
377 const LIST_NODE_T *node = &block->nodes[j];
378 WstListSlotState node_state = state[slot];
379 size_t target;
380
381 if (node->hdr.next != NULL) {
382 WstListGraphEdge kind = WST_LIST_GRAPH_EDGE_BAD;
383 if (list_dump_slot_of(list, node->hdr.next, &target) &&
384 target < total) {
385 if (node_state == WST_LIST_SLOT_FREE &&
386 state[target] == WST_LIST_SLOT_FREE)
387 kind = WST_LIST_GRAPH_EDGE_FREE_NEXT;
388 else if (node_state == WST_LIST_SLOT_LIVE &&
389 state[target] == WST_LIST_SLOT_LIVE)
390 kind = WST_LIST_GRAPH_EDGE_NEXT;
391 wst_list_graph_edge(graph, wst_list_graph_ref_slot(slot),
392 wst_list_graph_ref_slot(target), kind);
393 } else {
394 wst_list_graph_edge(graph, wst_list_graph_ref_slot(slot),
395 wst_list_graph_ref_null(),
396 WST_LIST_GRAPH_EDGE_BAD);
397 }
398 }
399 if (node->hdr.prev != NULL) {
400 WstListGraphEdge kind = WST_LIST_GRAPH_EDGE_BAD;
401 if (node_state == WST_LIST_SLOT_LIVE &&
402 list_dump_slot_of(list, node->hdr.prev, &target) &&
403 target < total && state[target] == WST_LIST_SLOT_LIVE)
404 kind = WST_LIST_GRAPH_EDGE_PREV;
405
406 if (list_dump_slot_of(list, node->hdr.prev, &target) &&
407 target < total)
408 wst_list_graph_edge(graph, wst_list_graph_ref_slot(slot),
409 wst_list_graph_ref_slot(target), kind);
410 else
411 wst_list_graph_edge(graph, wst_list_graph_ref_slot(slot),
412 wst_list_graph_ref_null(),
413 WST_LIST_GRAPH_EDGE_BAD);
414 }
415 }
416 wst_list_graph_block_end(graph);
417 base += block->cap;
418 }
419
420 WstListGraphRef header = wst_list_graph_ref_header();
421 WstListGraphRef null_ref = wst_list_graph_ref_null();
422 size_t target;
423
424 if (list->hdr.head == NULL) {
425 wst_list_graph_edge(graph, header, null_ref, WST_LIST_GRAPH_EDGE_HEAD);
426 } else if (list_dump_slot_of(list, list->hdr.head, &target) &&
427 target < total) {
428 WstListGraphEdge kind = state[target] == WST_LIST_SLOT_LIVE
429 ? WST_LIST_GRAPH_EDGE_HEAD
430 : WST_LIST_GRAPH_EDGE_BAD_HEAD;
431 wst_list_graph_edge(graph, header, wst_list_graph_ref_slot(target),
432 kind);
433 } else {
434 wst_list_graph_edge(graph, header, null_ref,
435 WST_LIST_GRAPH_EDGE_BAD_HEAD);
436 }
437
438 if (list->hdr.tail == NULL) {
439 wst_list_graph_edge(graph, header, null_ref, WST_LIST_GRAPH_EDGE_TAIL);
440 } else if (list_dump_slot_of(list, list->hdr.tail, &target) &&
441 target < total) {
442 WstListGraphEdge kind = state[target] == WST_LIST_SLOT_LIVE
443 ? WST_LIST_GRAPH_EDGE_TAIL
444 : WST_LIST_GRAPH_EDGE_BAD_TAIL;
445 wst_list_graph_edge(graph, header, wst_list_graph_ref_slot(target),
446 kind);
447 } else {
448 wst_list_graph_edge(graph, header, null_ref,
449 WST_LIST_GRAPH_EDGE_BAD_TAIL);
450 }
451
452 if (list->hdr.free == NULL) {
453 wst_list_graph_edge(graph, header, null_ref,
454 WST_LIST_GRAPH_EDGE_FREE_HEAD);
455 } else if (list_dump_slot_of(list, list->hdr.free, &target) &&
456 target < total) {
457 WstListGraphEdge kind = state[target] == WST_LIST_SLOT_FREE
458 ? WST_LIST_GRAPH_EDGE_FREE_HEAD
459 : WST_LIST_GRAPH_EDGE_BAD_FREE;
460 wst_list_graph_edge(graph, header, wst_list_graph_ref_slot(target),
461 kind);
462 } else {
463 wst_list_graph_edge(graph, header, null_ref,
464 WST_LIST_GRAPH_EDGE_BAD_FREE);
465 }
466
467 wst_list_graph_emit(graph, "linked list graph");
468 wst_list_graph_free(graph);
469 free(state);
