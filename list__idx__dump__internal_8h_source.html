13static uint32_t list_dump_len(const LIST_T *list)
14{
15 if (list->hdr.nodes == NULL)
16 return 0;
17
18 uint32_t len = list->hdr.len;
19 if ((size_t)len > list->hdr.cap)
20 len = (uint32_t)list->hdr.cap;
21 return len;
22}

24static void list_dump_states(const LIST_T *list, WstListSlotState *state,
25 uint32_t len)
26{
27 const LIST_NODE_T *nodes = (const LIST_NODE_T *)list->hdr.nodes;
28 if (nodes == NULL)
29 return;
30
31 uint32_t link = nodes[0].hdr.next;
32 for (uint32_t step = 0; link != 0 && link <= len && step <= len; step++) {
33 if (state[link] == WST_LIST_SLOT_ORPHAN)
34 state[link] = WST_LIST_SLOT_LIVE;
35 link = nodes[link].hdr.next;
36 }
37
38 link = list->hdr.free;
39 for (uint32_t step = 0; link != 0 && link <= len && step <= len; step++) {
40 if (state[link] == WST_LIST_SLOT_ORPHAN)
41 state[link] = WST_LIST_SLOT_FREE;
42 link = nodes[link].hdr.next;
43 }
44}

47static void list_dump_check_chain(const LIST_NODE_T *nodes, uint32_t start,
48 uint32_t len, const char *name,
49 WstListDiag *diag)
50{
51 uint32_t link = start;
52 for (uint32_t step = 0; link != 0; step++) {
53 if (link > len) {
54 wst_list_diag_add(diag, "%s chain leaves node array at %u", name,
55 link);
56 return;
57 }
58 if (step >= len) {
59 wst_list_diag_add(diag, "%s chain does not terminate", name);
60 return;
61 }
62 link = nodes[link].hdr.next;
63 }
64}

67static void list_dump_check_live(const LIST_NODE_T *nodes, uint32_t len,
68 uint32_t slot, uint32_t head, uint32_t tail,
69 const WstListSlotState *state,
70 WstListDiag *diag)
71{
72 uint32_t links[] = { nodes[slot].hdr.prev, nodes[slot].hdr.next };
73 const char *names[] = { "prev", "next" };
74 uint32_t ends[] = { head, tail };
75
76 for (int i = 0; i < 2; i++) {
77 uint32_t link = links[i];
78 if (link == 0) {
79 if (slot != ends[i])
80 wst_list_diag_add(diag, "live slot %u %s=0 but is not %s",
81 slot, names[i], i == 0 ? "head" : "tail");
82 continue;
83 }
84 if (link > len) {
85 wst_list_diag_add(diag, "live slot %u %s=%u is outside node array",
86 slot, names[i], link);
87 continue;
88 }
89 if (state[link] != WST_LIST_SLOT_LIVE) {
90 wst_list_diag_add(diag, "live slot %u %s -> %s slot %u", slot,
91 names[i],
92 wst_list_slot_state_name(state[link]), link);
93 continue;
94 }
95
96 uint32_t back =
97 i == 0 ? nodes[link].hdr.next : nodes[link].hdr.prev;
98 if (back != slot)
99 wst_list_diag_add(diag,
100 "live slot %u %s=%u but slot %u %s=%u", slot,
101 names[i], link, link,
102 i == 0 ? "next" : "prev", back);
103 }
104}

107static void list_dump_check_free(const LIST_NODE_T *nodes, uint32_t len,
108 uint32_t slot,
109 const WstListSlotState *state,
110 WstListDiag *diag)
111{
112 if (nodes[slot].hdr.prev != 0)
113 wst_list_diag_add(diag, "free slot %u prev=%u should be 0", slot,
114 nodes[slot].hdr.prev);
115
116 uint32_t next = nodes[slot].hdr.next;
117 if (next == 0)
118 return;
119
120 if (next > len) {
121 wst_list_diag_add(diag,
122 "free slot %u next=%u is outside node array", slot,
123 next);
124 } else if (state[next] != WST_LIST_SLOT_FREE) {
125 wst_list_diag_add(diag, "free slot %u next -> %s slot %u", slot,
126 wst_list_slot_state_name(state[next]), next);
127 }
128}

131static void list_dump_check(const LIST_T *list, const WstListSlotState *state,
132 uint32_t len, WstListDiag *diag)
133{
134 const LIST_NODE_T *nodes = (const LIST_NODE_T *)list->hdr.nodes;
135 if (nodes == NULL) {
136 wst_list_diag_add(diag, "node array is NULL");
137 return;
138 }
139 if (state == NULL) {
140 wst_list_diag_add(diag, "slot state allocation failed");
141 return;
142 }
143
144 uint32_t head = nodes[0].hdr.next;
145 uint32_t tail = nodes[0].hdr.prev;
146 uint32_t free = list->hdr.free;
147
148 if (head > len)
149 wst_list_diag_add(diag, "head=%u is past len %u", head, len);
150 if (tail > len)
151 wst_list_diag_add(diag, "tail=%u is past len %u", tail, len);
152 if (free > len)
153 wst_list_diag_add(diag, "free=%u is past len %u", free, len);
154 if ((head == 0) != (tail == 0))
155 wst_list_diag_add(diag, "head=%u but tail=%u", head, tail);
156
157 if (head != 0 && head <= len && state[head] != WST_LIST_SLOT_LIVE)
158 wst_list_diag_add(diag, "head -> %s slot %u",
159 wst_list_slot_state_name(state[head]), head);
160 if (tail != 0 && tail <= len && state[tail] != WST_LIST_SLOT_LIVE)
161 wst_list_diag_add(diag, "tail -> %s slot %u",
162 wst_list_slot_state_name(state[tail]), tail);
163 if (free != 0 && free <= len && state[free] != WST_LIST_SLOT_FREE)
164 wst_list_diag_add(diag, "free -> %s slot %u",
165 wst_list_slot_state_name(state[free]), free);
166
167 list_dump_check_chain(nodes, head, len, "live", diag);
168 list_dump_check_chain(nodes, free, len, "free", diag);
169
170 for (uint32_t slot = 1; slot <= len; slot++) {
171 switch (state[slot]) {
172 case WST_LIST_SLOT_LIVE:
173 list_dump_check_live(nodes, len, slot, head, tail, state, diag);
174 break;
175 case WST_LIST_SLOT_FREE:
176 list_dump_check_free(nodes, len, slot, state, diag);
177 break;
178 case WST_LIST_SLOT_ORPHAN:
179 default:
180 wst_list_diag_add(diag, "slot %u is neither live nor free", slot);
181 break;
182 }
183 }
184}

186static void list_dump_graph(const LIST_T *list, size_t issues)
187{
188 if (!wst_log_accepts_images())
189 return;
190
191 const LIST_NODE_T *nodes = (const LIST_NODE_T *)list->hdr.nodes;
192 if (nodes == NULL || list->hdr.cap == 0)
193 return;
194
195 uint32_t len = list_dump_len(list);
196 WstListSlotState *state = calloc((size_t)len + 1, sizeof(*state));
197 if (state == NULL)
198 return;
199
200 list_dump_states(list, state, len);
201 WstListSlotCounts counts = wst_list_slot_counts(state + 1, len);
202
203 WstListGraph *graph = wst_list_graph_new(
204 WST_LIST_STR(LIST_T),
205 "len=%u\ncap=%zu\nlive=%zu\npooled=%zu\n"
206 "orphan=%zu\nfree=%u\nissues=%zu",
207 len, list->hdr.cap, counts.live, counts.pooled, counts.orphan,
208 list->hdr.free, issues);
209 if (graph == NULL) {
210 free(state);
211 return;
212 }
213
214 for (uint32_t i = 1; i <= len; i++) {
215 const LIST_NODE_T *node = &nodes[i];
216 WstListSlotState node_state = state[i];
217
218#ifdef LIST_FMT
219 wst_list_graph_node(graph, i, node_state,
220 "#%u\nprev=%u\nnext=%u\ndata=" LIST_FMT, i,
221 node->hdr.prev, node->hdr.next, node->data);
222#else
223 wst_list_graph_node(graph, i, node_state,
224 "#%u\nprev=%u\nnext=%u\ndata=%p", i,
225 node->hdr.prev, node->hdr.next,
226 (const void *)&node->data);
227#endif
228
229 uint32_t links[] = { node->hdr.prev, node->hdr.next };
230 WstListGraphEdge good[] = { WST_LIST_GRAPH_EDGE_PREV,
231 WST_LIST_GRAPH_EDGE_NEXT };
232 for (int k = 0; k < 2; k++) {
233 uint32_t link = links[k];
234 if (link == 0)
235 continue;
236
237 WstListGraphEdge kind = WST_LIST_GRAPH_EDGE_BAD;
238 if (link <= len) {
239 bool same = node_state == WST_LIST_SLOT_LIVE &&
240 state[link] == WST_LIST_SLOT_LIVE;
241 if (k == 1 && node_state == WST_LIST_SLOT_FREE &&
242 state[link] == WST_LIST_SLOT_FREE)
243 same = true;
244 if (same)
245 kind = good[k];
246 wst_list_graph_edge(graph, wst_list_graph_ref_slot(i),
247 wst_list_graph_ref_slot(link), kind);
248 } else {
249 wst_list_graph_edge(graph, wst_list_graph_ref_slot(i),
250 wst_list_graph_ref_null(), kind);
251 }
252 }
253 }
254
255 WstListGraphRef header = wst_list_graph_ref_header();
256 WstListGraphRef null_ref = wst_list_graph_ref_null();
257 uint32_t ends[] = { nodes[0].hdr.next, nodes[0].hdr.prev,
258 list->hdr.free };
259 WstListGraphEdge good[] = { WST_LIST_GRAPH_EDGE_HEAD,
260 WST_LIST_GRAPH_EDGE_TAIL,
261 WST_LIST_GRAPH_EDGE_FREE_HEAD };
262 WstListGraphEdge bad[] = { WST_LIST_GRAPH_EDGE_BAD_HEAD,
263 WST_LIST_GRAPH_EDGE_BAD_TAIL,
264 WST_LIST_GRAPH_EDGE_BAD_FREE };
265 WstListSlotState want[] = { WST_LIST_SLOT_LIVE, WST_LIST_SLOT_LIVE,
266 WST_LIST_SLOT_FREE };
267
268 for (int i = 0; i < 3; i++) {
269 if (ends[i] == 0) {
270 wst_list_graph_edge(graph, header, null_ref, good[i]);
271 } else if (ends[i] <= len) {
272 WstListGraphEdge kind = state[ends[i]] == want[i] ? good[i]
273 : bad[i];
274 wst_list_graph_edge(graph, header,
275 wst_list_graph_ref_slot(ends[i]), kind);
276 } else {
277 wst_list_graph_edge(graph, header, null_ref, bad[i]);
278 }
279 }
280
281 wst_list_graph_emit(graph, "linked list graph");
282 wst_list_graph_free(graph);
283 free(state);
284}
