373static WstListErr list_drop_blocks(LIST_T *list)
374{
375 assert(list != NULL);
376
377 size_t count = list->blocks.length;
378 LOG_D("dropping %zu node blocks", count);
379
380 for (size_t i = 0; i < count; i++) {
381 LIST_BLOCK_T *block = list->blocks.data[i];
382 LOG_V("freeing block %zu/%zu at %p (%zu nodes)", i + 1, count,
383 (const void *)block, block->cap);
384 free(block);
385 }
386
387 WstVecErr verr = BLOCK_VEC_F(free)(&list->blocks);
388 if (verr != WST_VEC_NO_ERR) {
389 LOG_E("cannot free block vector: %s", wst_vec_err_str(verr));
390 return wst_list_err_from_vec(verr);
391 }
392
393 LOG_D("dropped node blocks and block vector");
394 return WST_LIST_NO_ERR;
395}

398static WstListErr list_new_block(LIST_T *list)
399{
400 assert(list != NULL);
401
402 size_t cap = LIST_BLOCK_INIT_CAP;
403 if (list->blocks.length > 0) {
404 size_t prev_cap = list->blocks.data[list->blocks.length - 1]->cap;
405 if (prev_cap > SIZE_MAX / 2) {
406 LOG_E("block capacity %zu overflows size_t", prev_cap);
407 return WST_LIST_ERR_OVERFLOW;
408 }
409 cap = prev_cap * 2;
410 }
411
412 if (cap == 0)
413 cap = 1;
414
415 if (cap > (SIZE_MAX - sizeof(LIST_BLOCK_T)) / sizeof(LIST_NODE_T)) {
416 LOG_E("block capacity %zu overflows size_t", cap);
417 return WST_LIST_ERR_OVERFLOW;
418 }
419
420 LIST_BLOCK_T *block = malloc(sizeof(LIST_BLOCK_T) +
421 cap * sizeof(LIST_NODE_T));
422 if (block == NULL) {
423 LOG_E("cannot allocate %zu-node block: %s", cap, strerror(errno));
424 return WST_LIST_ERR_NOMEM;
425 }
426 block->cap = cap;
427
428 LIST_BLOCK_PTR_T block_ptr = block;
429 WstVecErr verr = BLOCK_VEC_F(push)(&list->blocks, &block_ptr);
430 if (verr != WST_VEC_NO_ERR) {
431 free(block);
432 return wst_list_err_from_vec(verr);
433 }
434
435 wst_list_hdr_add_free(&list->hdr, block->nodes, block->cap,
436 sizeof(LIST_NODE_T));
437 LOG_D("allocated block %zu with %zu nodes at %p",
438 list->blocks.length, block->cap, (const void *)block);
439 return WST_LIST_NO_ERR;
440}

443static WstListErr list_alloc_node(LIST_T *list, LIST_NODE_T **out)
444{
445 assert(list != NULL);
446 assert(out != NULL);
447
448 WstListNodeHdr *hdr = wst_list_hdr_take_free(&list->hdr);
449 if (hdr == NULL) {
450 WstListErr err = list_new_block(list);
451 if (err != WST_LIST_NO_ERR)
452 return err;
453
454 hdr = wst_list_hdr_take_free(&list->hdr);
455 assert(hdr != NULL);
456 }
457
458 *out = (LIST_NODE_T *)hdr;
459 return WST_LIST_NO_ERR;

511WstListErr list_free(LIST_T *list)
512{
513 assert(list != NULL);
514
515 WstListErr err = list_verify(list);
516 if (err != WST_LIST_NO_ERR)
517 return err;
518
519 LOG_D("freeing list with %zu node blocks", list->blocks.length);
520
521 err = list_drop_blocks(list);
522 if (err != WST_LIST_NO_ERR)
523 return err;
524
525 wst_list_hdr_reset(&list->hdr);
526 LOG_D("freed list");
527 return WST_LIST_NO_ERR;
528}

530WstListErr list_clear(LIST_T *list)
531{
532 assert(list != NULL);
533
534 WstListErr err = list_verify(list);
535 if (err != WST_LIST_NO_ERR)
536 return err;
537
538 wst_list_hdr_reset(&list->hdr);
539
540 for (size_t i = 0; i < list->blocks.length; i++) {
541 LIST_BLOCK_T *block = list->blocks.data[i];
542 LOG_V("pooling block %zu/%zu at %p (%zu nodes)", i + 1,
543 list->blocks.length, (const void *)block, block->cap);
544 wst_list_hdr_add_free(&list->hdr, block->nodes, block->cap,
545 sizeof(LIST_NODE_T));
546 }
547
548 LOG_D("cleared list, %zu blocks pooled", list->blocks.length);
549 return WST_LIST_NO_ERR;
550}

585void list_dump(const LIST_T *list)
586{
587 LOG_V("Dumping list at %p", (const void *)list);
588
589 if (list == NULL) {
590 fprintf(stderr, "list: <null>\n");
591 return;
592 }
593
594 fprintf(stderr, "list <%p>:\n"
595 " head: %p\n"
596 " tail: %p\n"
597 " free: %p\n",
598 (const void *)list, (const void *)list->hdr.head,
599 (const void *)list->hdr.tail, (const void *)list->hdr.free);
600
601 bool sane = list_verify(list) == WST_LIST_NO_ERR;
602
603#ifdef LIST_FMT
604 if (sane) {
605 fprintf(stderr, " elements:\n");
606 size_t i = 0;
607 for (const WstListNodeHdr *hdr = list->hdr.head; hdr != NULL;
608 hdr = hdr->next, i++)
609 fprintf(stderr, " [%4zu]: " LIST_FMT "\n", i,
610 ((const LIST_NODE_T *)hdr)->data);
611 } else {
612 fprintf(stderr, " elements: omitted due to verification failure\n");
613 }
614#endif
615
616 if (sane) {
617 fprintf(stderr, " blocks (%zu):\n", list->blocks.length);
618 for (size_t i = 0; i < list->blocks.length; i++)
619 fprintf(stderr, " [%4zu]: %zu nodes\n", i,
620 list->blocks.data[i]->cap);
621 } else {
622 fprintf(stderr, " blocks: omitted due to verification failure\n");
623 }
624
625 BLOCK_VEC_F(dump)(&list->blocks);
626}
